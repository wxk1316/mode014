cmake_minimum_required(VERSION 3.18)
project(campus_centerpoint LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 87)
endif()
find_package(CUDAToolkit REQUIRED)
find_path(TENSORRT_INCLUDE NvInfer.h PATHS /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu /usr/local/TensorRT/include)
find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
find_library(TENSORRT_PARSER nvonnxparser REQUIRED)
if(NOT TENSORRT_INCLUDE)
  message(FATAL_ERROR "TensorRT development headers missing (libnvinfer-dev/libnvonnxparsers-dev)")
endif()
add_library(campus_centerpoint SHARED centerpoint.cu)
target_include_directories(campus_centerpoint PRIVATE ${TENSORRT_INCLUDE})
target_link_libraries(campus_centerpoint PRIVATE CUDA::cudart ${TENSORRT_LIBRARY} ${TENSORRT_PARSER})
target_compile_options(campus_centerpoint PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
set_target_properties(campus_centerpoint PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
install(TARGETS campus_centerpoint LIBRARY DESTINATION campus_lidar_perception/lib)
